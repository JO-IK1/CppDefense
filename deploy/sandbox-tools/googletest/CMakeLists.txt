cmake_minimum_required(VERSION 3.20)
project(CppDefensePrebuiltGoogleTest LANGUAGES CXX)
find_package(GTest CONFIG REQUIRED)
foreach(name gtest gtest_main gmock gmock_main)
  if(TARGET GTest::${name})
    set_property(TARGET GTest::${name} PROPERTY IMPORTED_GLOBAL TRUE)
  endif()
  if(TARGET GTest::${name} AND NOT TARGET ${name})
    add_library(${name} ALIAS GTest::${name})
  endif()
endforeach()
# Compatibility for projects which manually refer to gtest_SOURCE_DIR/include.
set(gtest_SOURCE_DIR /usr/src/googletest/googletest PARENT_SCOPE)
set(gmock_SOURCE_DIR /usr/src/googletest/googlemock PARENT_SCOPE)
